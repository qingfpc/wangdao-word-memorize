    <resultMap id="vocabularyInfoMap" type="com.cskaoyan.wordmemorize.dao.entity.VocDO" autoMapping="true">
        <id column="id" property="id"/>
        <collection property="interpretations" ofType="com.cskaoyan.wordmemorize.dao.entity.VocMeaningDO"
                    select="getVocMeaning" column="id">
        </collection>

        </collection>
    </resultMap>
    <select id="getVocabularyInfo" resultMap="vocabularyInfoMap">
        <choose>
            <when test="bookId != null">
                select
                <include refid="voc_columns"/>
                from voc_chap_book vcb inner join voc v on v.id=vcb.voc_id and vcb.book_id=#{bookId}
                <where>
                    v.is_deleted=0 and vcb.is_deleted=0
                    <if test="chapterId != null">
                        and vcb.chapter_id=#{chapterId}
                    </if>
                    <if test="keyword != null and keyword != ''">
                        and v.content LIKE CONCAT('%', #{keyword}, '%')
                    </if>
                </where>
            </when>
            <otherwise>
                select
                <include refid="voc_columns"/>
                from voc v
                where is_deleted=0
                <if test="keyword != null and keyword != ''">
                    and v.content LIKE CONCAT('%', #{keyword}, '%')
                </if>
            </otherwise>
        </choose>
    </select>

    <select id="getVocMeaning" resultType="com.cskaoyan.wordmemorize.dao.entity.VocMeaningDO">
        select
        <include refid="voc_meaning_columns"/>
        from voc_meaning vm where vm.voc_id=#{id}
    </select>

    <select id="getVocExampleSentence" resultType="com.cskaoyan.wordmemorize.dao.entity.ExampleSentenceDO">
        select
        <include refid="voc_example_sentence_columns"/>
        from example_sentence es where es.voc_id=#{id}
    </select>

    <select id="getVocDetail" resultMap="vocDetailMap">
        select v.id, v.content, v.uk_phonetic, v.us_phonetic, v.us_pronounciation, v.uk_pronounciation, vm.id as vm_id, vm.voc_id, vm.type, vm.inter_content
             ,es.id as es_id, es.voc_id, es.content as es_content, es.meaning, if(uvr.voc_id is null, 0, 1) reviewed, if(unr.voc_id is null, 0, 1) not_remember
        from voc v
        inner join voc_meaning vm on vm.voc_id=v.id and v.is_deleted=0 and vm.is_deleted=0
        inner join example_sentence es on es.voc_id=v.id and es.is_deleted=0
        left join user_voc_rev uvr on uvr.user_id=#{userId} and uvr.voc_id=v.id and uvr.is_deleted=0
        left join user_not_remember unr on unr.user_id=#{userId} and unr.voc_id=v.id and unr.is_deleted=0
        where v.id=#{vocId}
    </select>
   
   Mapper方法: 
   VocDO getVocDetail(@Param("userId") Long userId, @Param("vocId") Long vocId);
   这里需要注意的是, 对于单词详情来说，我们还需要查询出该单词是否已经被收藏，因为以上sql连接的表太多，所以并未放在一个sql语句中，我们可以在
   到user_favorite表中在在根据单词id查询该单词是否已被收藏
   
   service层方法：
    @Override
    public VocDTO getVocDetail(Long vocId, Long userId) {

        VocDO vocDetail = vocMapper.getVocDetail(userId, vocId);
        List<UserVocExampleSentenceDO> userVocSentence = vocMapper.getUserVocExampleSentence(userId, vocId);
        vocDetail.setUserExamples(userVocSentence);

        LambdaQueryWrapper<UserFavoriteDO> favoriteWrapper = Wrappers.lambdaQuery(UserFavoriteDO.class)
                .eq(UserFavoriteDO::getUserId, userId)
                .eq(UserFavoriteDO::getVocId, vocId);
        // 获取用户是否收藏
        Long count = userFavoriteMapper.selectCount(favoriteWrapper);
        vocDetail.setFavorite(count > 0 ? 1 : 0);
        return vocConverter.vocDO2DTO(vocDetail);
    }
